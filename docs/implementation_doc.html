import numpy as np

class RLSFilter:
    def __init__(self, num_taps=16, lambda_factor=0.99):
        self.n = num_taps
        self.lam = lambda_factor
        self.w = np.zeros(num_taps)
        self.P = np.eye(num_taps) / 0.01

    def filter(self, d, x_vec):
        # d: desired signal + noise, x_vec: noise reference
        k = (self.P @ x_vec) / (self.lam + x_vec.T @ self.P @ x_vec)
        e = d - self.w.T @ x_vec
        self.w += k * e
        self.P = (self.P - np.outer(k, x_vec.T @ self.P)) / self.lam
        return e
